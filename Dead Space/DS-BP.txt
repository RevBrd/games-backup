      <div style={{display:"flex",gap:12,justifyContent:"center"}}>
        <button onClick={onRestart} style={{background:"linear-gradient(to bottom,#55ff55,#22aa22)",border:"3px outset #44dd44",borderRadius:6,padding:"10px 24px",fontSize:16,fontWeight:900,color:"#003300",cursor:"pointer",letterSpacing:2,fontFamily:"Impact,'Arial Black',sans-serif"}}><RotateCcw size={14} style={{marginRight:6,verticalAlign:-2}}/> PLAY AGAIN</button>
        <button onClick={onMenu} style={{background:"linear-gradient(to bottom,#6688bb,#445577)",border:"3px outset #7799cc",borderRadius:6,padding:"10px 24px",fontSize:16,fontWeight:900,color:"#ddeeff",cursor:"pointer",letterSpacing:2,fontFamily:"Impact,'Arial Black',sans-serif"}}><Home size={14} style={{marginRight:6,verticalAlign:-2}}/> MENU</button>

// ======== MAIN GAME ========
export default function DeadSpaceBlockParty(){
  const [screen,setScreen]=useState("splash");
  const [board,setBoard]=useState(createBoard);
  const [current,setCurrent]=useState(null);
  const [currentType,setCurrentType]=useState(null);
  const [curPos,setCurPos]=useState({r:0,c:0});
  const [nextType,setNextType]=useState(null);
  const [score,setScore]=useState(0);
  const [lines,setLines]=useState(0);
  const [level,setLevel]=useState(0);
  const [paused,setPaused]=useState(false);
  const [won,setWon]=useState(false);
  const [unlockedCrew,setUnlockedCrew]=useState(()=>CREW.filter(c=>c.id==="isaac"||c.id==="necro"));
  const [markers,setMarkers]=useState(0);
  const [markerCells,setMarkerCells]=useState(()=>new Set());
  const [showMarkerCollect,setShowMarkerCollect]=useState(false);
  const [transCards,setTransCards]=useState([]);
  const [encIdx,setEncIdx]=useState(0);

  const [tipVisible,setTipVisible]=useState(false);
  const [tipIndex,setTipIndex]=useState(0);
  const tipAutoTimer=useRef(null);const tipGapTimer=useRef(null);const tipUsed=useRef(new Set());

  const tickRef=useRef(null);const boardRef=useRef(board);const currentRef=useRef(current);
  const ctRef=useRef(currentType);const cpRef=useRef(curPos);const pausedRef=useRef(paused);
  const goRef=useRef(false);const lvRef=useRef(level);const ucRef=useRef(unlockedCrew);
  const mcRef=useRef(markerCells);const markersRef=useRef(0);

  boardRef.current=board;currentRef.current=current;ctRef.current=currentType;
  cpRef.current=curPos;pausedRef.current=paused;lvRef.current=level;
  ucRef.current=unlockedCrew;mcRef.current=markerCells;markersRef.current=markers;

  const speed=Math.max(100,800-level*63);
  const bagRef=useRef(shuffleBag());
  const pull=useCallback(()=>{if(!bagRef.current.length)bagRef.current=shuffleBag();return bagRef.current.pop()},[]);
  const actualNextRef=useRef(null);
  const transiting=transCards.length>0;

  const spawn=useCallback((boardState,overrideNext)=>{
    const type=overrideNext||actualNextRef.current||pull();
    const shape=SHAPES[type];const col=Math.floor((COLS-shape[0].length)/2);
    if(collides(boardState,shape,0,col)){goRef.current=true;setScreen("gameover");return}
    setCurrent(shape);setCurrentType(type);ctRef.current=type;setCurPos({r:0,c:col});
    const realNext=pull();actualNextRef.current=realNext;
    const{displayDrift}=getMarkerInterference(markersRef.current);
    if(Math.random()<displayDrift){const f=TYPES.filter(t=>t!==realNext);setNextType(f[Math.floor(Math.random()*f.length)])}
    else setNextType(realNext);
  },[pull]);

  const buildTransition=useCallback((completedIdx)=>{
    const lvl=LEVELS[completedIdx];if(!lvl)return[];
    const cards=[];
    if(lvl.unlocks.length>0){
      const members=lvl.unlocks.map(id=>CREW.find(c=>c.id===id)).filter(Boolean);
      if(members.length)cards.push({type:'unlock',members,completionChar:lvl.completion.char,completionText:lvl.completion.text,name:lvl.name});
    } else {
      cards.push({type:'complete',char:lvl.completion.char,text:lvl.completion.text,name:lvl.name});
    }
    if(lvl.ending){
      cards.push({type:'quote',char:'nicole',text:"We're bringing the Marker home safe and sound, and it's all thanks to you! ✦"});
      cards.push({type:'finale'});
    } else {
      const nextLvl=LEVELS[completedIdx+1];
      if(nextLvl)cards.push({type:'intro',char:nextLvl.intro.char,text:nextLvl.intro.text,levelNum:completedIdx+2,name:nextLvl.name});
    }
    return cards;
  },[]);

  const lock=useCallback(()=>{
    const b=boardRef.current.map(r=>[...r]);const shape=currentRef.current;
    const{r:pr,c:pc}=cpRef.current;const ct=ctRef.current;
    if(!shape)return b;
    for(let r=0;r<shape.length;r++)for(let c=0;c<shape[r].length;c++)
      if(shape[r][c]&&pr+r>=0)b[pr+r][pc+c]=PIECE_COLORS[ct]||"#fff";
    let cleared=0;const clearedRows=[];const newBoard=[];
    for(let r=0;r<ROWS;r++){if(b[r].every(cell=>cell!==null)){cleared++;clearedRows.push(r)}else newBoard.push(b[r])}
    while(newBoard.length<ROWS)newBoard.unshift(Array(COLS).fill(null));
    let markersCollected=0;const mc=new Set(mcRef.current);
    for(const row of clearedRows)for(let c=0;c<COLS;c++)if(mc.has(`${row},${c}`)){markersCollected++;mc.delete(`${row},${c}`)}
    const newMc=new Set();
    for(const key of mc){const[mr,mcol]=key.split(",").map(Number);const shift=clearedRows.filter(cr=>cr>mr).length;newMc.add(`${mr+shift},${mcol}`)}
    if(cleared>0&&newMc.size<2&&Math.random()<0.2){
      const spots=[];for(let r=8;r<ROWS-2;r++)for(let c=0;c<COLS;c++)if(!newBoard[r][c]&&!newMc.has(`${r},${c}`))spots.push(`${r},${c}`);
      if(spots.length)newMc.add(spots[Math.floor(Math.random()*spots.length)]);
    }
    setMarkerCells(newMc);mcRef.current=newMc;
    if(markersCollected>0){setMarkers(m=>m+markersCollected);setScore(s=>s+markersCollected*250);setShowMarkerCollect(true);setTimeout(()=>setShowMarkerCollect(false),1500)}
    if(cleared>0){
      const lv=lvRef.current;const pts=[0,100,300,500,800][cleared]*(lv+1);setScore(s=>s+pts);
      setLines(l=>{
        const nl=l+cleared;const newLv=Math.floor(nl/10);
        if(newLv!==lv){
          setLevel(newLv);
          const completedIdx=newLv-1;
          const lvlData=LEVELS[completedIdx];
          if(lvlData){
            // Unlock crew immediately
            if(lvlData.unlocks.length>0){
              const newMembers=lvlData.unlocks.map(id=>CREW.find(c=>c.id===id)).filter(Boolean);
              setUnlockedCrew(prev=>{const next=[...prev,...newMembers.filter(m=>!prev.find(p=>p.id===m.id))];ucRef.current=next;return next});
            }
            const cards=buildTransition(completedIdx);
            setTransCards(cards);
            if(lvlData.ending){goRef.current=true}
          }
        }
        return nl;
      });
    }
    setBoard(newBoard);return newBoard;
  },[buildTransition]);

  const dismissCard=useCallback(()=>{
    setTransCards(prev=>{
      const next=prev.slice(1);
      if(prev[0]?.type==='finale'){setScreen("gameover");setWon(true);return []}
      return next;
    });
  },[]);

  useEffect(()=>{
    if(screen!=="playing"||paused||transiting){clearInterval(tickRef.current);return}
    tickRef.current=setInterval(()=>{
      if(pausedRef.current||goRef.current)return;
      const shape=currentRef.current;const{r,c}=cpRef.current;if(!shape)return;
      if(!collides(boardRef.current,shape,r+1,c))setCurPos(p=>({...p,r:p.r+1}));
      else{const nb=lock();spawn(nb)}
    },speed);
    return()=>clearInterval(tickRef.current);
  },[screen,paused,speed,lock,spawn,transiting]);

  useEffect(()=>{
    if(screen!=="playing")return;
    const h=e=>{
      if(goRef.current||transiting)return;
      if(e.key==="p"||e.key==="P"){setPaused(p=>!p);return}
      if(pausedRef.current)return;
      const shape=currentRef.current;const{r,c}=cpRef.current;if(!shape)return;
      let key=e.key;
      if(key.startsWith("Arrow")){
        const{inputLoss,inputDrift}=getMarkerInterference(markersRef.current);
        if(Math.random()<inputLoss){e.preventDefault();return}
        if(Math.random()<inputDrift){if(key==="ArrowLeft")key="ArrowRight";else if(key==="ArrowRight")key="ArrowLeft";else if(key==="ArrowUp"){e.preventDefault();return}}
      }
      if(key==="ArrowLeft"){e.preventDefault();if(!collides(boardRef.current,shape,r,c-1))setCurPos(p=>({...p,c:p.c-1}))}
      else if(key==="ArrowRight"){e.preventDefault();if(!collides(boardRef.current,shape,r,c+1))setCurPos(p=>({...p,c:p.c+1}))}
      else if(key==="ArrowDown"){e.preventDefault();if(!collides(boardRef.current,shape,r+1,c))setCurPos(p=>({...p,r:p.r+1}))}
      else if(key==="ArrowUp"){e.preventDefault();const rot=rotate(shape);if(!collides(boardRef.current,rot,r,c))setCurrent(rot);else if(!collides(boardRef.current,rot,r,c-1)){setCurrent(rot);setCurPos(p=>({...p,c:p.c-1}))}else if(!collides(boardRef.current,rot,r,c+1)){setCurrent(rot);setCurPos(p=>({...p,c:p.c+1}))}}
      else if(key===" "){e.preventDefault();let nr=r;while(!collides(boardRef.current,shape,nr+1,c))nr++;setCurPos({r:nr,c});setTimeout(()=>{const nb=lock();spawn(nb)},30)}
    };
    window.addEventListener("keydown",h);return()=>window.removeEventListener("keydown",h);
  },[screen,lock,spawn,transiting]);

  // Tips
  const availTips=useMemo(()=>{const ids=new Set(unlockedCrew.map(c=>c.id));return TIPS.map((t,i)=>({...t,idx:i})).filter(t=>ids.has(t.char))},[unlockedCrew]);
  const showNextTip=useCallback(()=>{if(goRef.current)return;const pool=availTips;if(!pool.length)return;if(tipUsed.current.size>=pool.length)tipUsed.current.clear();let pick;do{pick=pool[Math.floor(Math.random()*pool.length)]}while(tipUsed.current.has(pick.idx)&&tipUsed.current.size<pool.length);tipUsed.current.add(pick.idx);setTipIndex(pick.idx);setTipVisible(true);clearTimeout(tipAutoTimer.current);tipAutoTimer.current=setTimeout(()=>{setTipVisible(false);clearTimeout(tipGapTimer.current);tipGapTimer.current=setTimeout(()=>showNextTip(),15000)},15000)},[availTips]);
  const closeTip=useCallback(()=>{setTipVisible(false);clearTimeout(tipAutoTimer.current);clearTimeout(tipGapTimer.current);tipGapTimer.current=setTimeout(()=>showNextTip(),30000)},[showNextTip]);
  useEffect(()=>{if(screen==="playing"){tipUsed.current.clear();const t=setTimeout(()=>showNextTip(),5000);return()=>{clearTimeout(t);clearTimeout(tipAutoTimer.current);clearTimeout(tipGapTimer.current)}}},[screen,showNextTip]);

  // Encouragement
  const nextToUnlock=useMemo(()=>{for(const lvl of LEVELS){for(const id of lvl.unlocks){if(!unlockedCrew.find(c=>c.id===id))return CREW.find(c=>c.id===id)}}return null},[unlockedCrew]);
  useEffect(()=>{if(!nextToUnlock||screen!=="playing")return;const iv=setInterval(()=>setEncIdx(i=>i+1),8000);return()=>clearInterval(iv)},[nextToUnlock,screen]);

  const startPlaying=useCallback(()=>{
    const b=createBoard();setBoard(b);setScore(0);setLines(0);setLevel(0);
    goRef.current=false;setPaused(false);setTipVisible(false);setWon(false);
    setMarkers(0);setMarkerCells(new Set());setShowMarkerCollect(false);setTransCards([]);
    setUnlockedCrew(CREW.filter(c=>c.id==="isaac"||c.id==="necro"));ucRef.current=CREW.filter(c=>c.id==="isaac"||c.id==="necro");
    bagRef.current=shuffleBag();const first=bagRef.current.pop();const realNext=bagRef.current.pop();
    actualNextRef.current=realNext;setNextType(realNext);
    const shape=SHAPES[first];const col=Math.floor((COLS-shape[0].length)/2);
    setCurrent(shape);setCurrentType(first);ctRef.current=first;setCurPos({r:0,c:col});
    setScreen("playing");
  },[]);

  const ghostRow=useMemo(()=>{if(!current)return 0;let r=curPos.r;while(!collides(board,current,r+1,curPos.c))r++;return r},[board,current,curPos]);

  if(screen==="splash") return (<SplashScreen onStart={()=>setScreen("intro")}/>);
  if(screen==="intro") return (<IntroScreen onDone={startPlaying}/>);
  if(screen==="gameover") return (<EndScreen score={score} lines={lines} level={level} unlockedCrew={unlockedCrew} markers={markers} won={won} onRestart={startPlaying} onMenu={()=>setScreen("splash")}/>);

  const boardW=COLS*CELL,boardH=ROWS*CELL;
  const currentLevelData=LEVELS[Math.min(level,LEVELS.length-1)];
  const encMsgs=nextToUnlock&&ENCOURAGEMENT[nextToUnlock.id];
  const encMsg=encMsgs?encMsgs[encIdx%encMsgs.length]:null;
  const NextFace=nextToUnlock?FACES[nextToUnlock.id]:null;

  return(<div style={{width:"100%",minHeight:"100vh",background:"linear-gradient(180deg,#0a0020,#120830 50%,#0d1b3e)",display:"flex",flexDirection:"column",alignItems:"center",fontFamily:"'Segoe UI',Tahoma,sans-serif",position:"relative",overflow:"hidden"}}><Stars/>
    <div style={{marginTop:10,padding:"4px 20px",background:"rgba(0,100,80,.3)",border:"1px solid #00ff8855",borderRadius:4,zIndex:2,fontSize:10,color:"#00ff88",letterSpacing:2,fontFamily:"monospace"}}>CEC ENTERTAINMENT SUITE v2.1 — DEAD SPACE: BLOCK PARTY</div>
    <div style={{display:"flex",gap:20,marginTop:12,position:"relative",zIndex:2}}>
      <div style={{position:"relative"}}>
        <div style={{width:boardW,height:boardH,border:"2px solid #00ff8866",borderRadius:4,background:"rgba(0,10,20,.85)",position:"relative",boxShadow:"0 0 20px rgba(0,255,136,.1),inset 0 0 30px rgba(0,0,0,.5)"}}>
          <svg width={boardW} height={boardH} style={{position:"absolute",top:0,left:0,opacity:.08}}>{Array.from({length:COLS+1},(_,i)=><line key={`v${i}`} x1={i*CELL} y1={0} x2={i*CELL} y2={boardH} stroke="#00ff88"/>)}{Array.from({length:ROWS+1},(_,i)=><line key={`h${i}`} x1={0} y1={i*CELL} x2={boardW} y2={i*CELL} stroke="#00ff88"/>)}</svg>
          {board.map((row,r)=>row.map((cell,c)=>cell?<div key={`${r}-${c}`} style={{position:"absolute",left:c*CELL+1,top:r*CELL+1,width:CELL-2,height:CELL-2,borderRadius:3,background:cell,boxShadow:`inset 0 0 6px rgba(255,255,255,.3),0 0 4px ${cell}44`,border:"1px solid rgba(255,255,255,.15)"}}/>:null))}
          {[...markerCells].map(key=>{const[mr,mcc]=key.split(",").map(Number);return <div key={`m${key}`} style={{position:"absolute",left:mcc*CELL,top:mr*CELL,width:CELL,height:CELL,display:"flex",alignItems:"center",justifyContent:"center",fontSize:18,color:"#ffd700",textShadow:"0 0 8px rgba(255,215,0,.6)",pointerEvents:"none",zIndex:5,animation:"tw 1.5s ease-in-out infinite alternate"}}>✦</div>})}
          {current&&current.map((row,r)=>row.map((cell,c)=>cell?<div key={`g${r}-${c}`} style={{position:"absolute",left:(curPos.c+c)*CELL+1,top:(ghostRow+r)*CELL+1,width:CELL-2,height:CELL-2,borderRadius:3,border:`1px dashed ${PIECE_COLORS[currentType]}55`,background:`${PIECE_COLORS[currentType]}15`}}/>:null))}
          {current&&current.map((row,r)=>row.map((cell,c)=>cell&&curPos.r+r>=0?<div key={`p${r}-${c}`} style={{position:"absolute",left:(curPos.c+c)*CELL+1,top:(curPos.r+r)*CELL+1,width:CELL-2,height:CELL-2,borderRadius:3,background:PIECE_COLORS[currentType],boxShadow:`inset 0 0 6px rgba(255,255,255,.3),0 0 8px ${PIECE_COLORS[currentType]}66`,border:"1px solid rgba(255,255,255,.2)"}}/>:null))}
        </div>
        {paused&&<div style={{position:"absolute",inset:0,background:"rgba(0,0,0,.7)",display:"flex",flexDirection:"column",alignItems:"center",justifyContent:"center",borderRadius:4,zIndex:50}}>
          <div style={{color:"#ffd700",fontSize:28,fontWeight:900,fontFamily:"'Comic Sans MS',cursive"}}>PAUSED</div>
          <div style={{fontSize:12,color:"#aab",marginTop:8}}>Press P to resume</div>
          <button onClick={()=>{setPaused(false);setScreen("splash")}} style={{marginTop:16,background:"linear-gradient(to bottom,#6688bb,#445577)",border:"2px outset #7799cc",borderRadius:4,padding:"6px 16px",fontSize:12,fontWeight:700,color:"#ddeeff",cursor:"pointer",letterSpacing:1}}><Home size={12} style={{marginRight:4,verticalAlign:-2}}/> Main Menu</button>
        </div>}
        {transCards.length>0&&!paused&&<TransitionOverlay card={transCards[0]} onDismiss={dismissCard}/>}
        {showMarkerCollect&&<MarkerCollectPopup/>}
      </div>

        <div style={{background:"rgba(0,40,30,.6)",border:"1px solid #00ff8844",borderRadius:6,padding:"10px 12px"}}><div style={{fontSize:9,color:"#00ff88",letterSpacing:2,marginBottom:2,fontFamily:"monospace"}}>SCORE</div><div style={{fontSize:22,fontWeight:900,color:"#ffd700",fontFamily:"monospace"}}>{score}</div></div>
        <div style={{background:"rgba(0,40,30,.6)",border:"1px solid #00ff8844",borderRadius:6,padding:"8px 12px"}}><div style={{fontSize:9,color:"#00ff88",letterSpacing:2,marginBottom:2,fontFamily:"monospace"}}>LINES</div><div style={{fontSize:18,fontWeight:700,color:"#00e676",fontFamily:"monospace"}}>{lines}</div></div>
        <div style={{background:"rgba(0,40,30,.6)",border:"1px solid #00ff8844",borderRadius:6,padding:"8px 12px"}}><div style={{fontSize:9,color:"#00ff88",letterSpacing:2,marginBottom:2,fontFamily:"monospace"}}>LEVEL {level+1}</div><div style={{fontSize:11,color:"#88aacc",marginTop:2,lineHeight:1.3,fontStyle:"italic"}}>{currentLevelData?.name||"Deep Space"}</div></div>
        <div style={{background:"rgba(0,40,30,.6)",border:"1px solid #00ff8844",borderRadius:6,padding:"8px 12px",textAlign:"center"}}><div style={{fontSize:9,color:"#00ff88",letterSpacing:2,marginBottom:6,fontFamily:"monospace"}}>NEXT</div><div style={{display:"flex",justifyContent:"center"}}><PiecePreview type={nextType}/></div></div>
        <div style={{background:"rgba(0,40,30,.6)",border:"1px solid #ffd70044",borderRadius:6,padding:"8px 12px",textAlign:"center"}}><div style={{fontSize:9,color:"#ffd700",letterSpacing:2,marginBottom:2,fontFamily:"monospace"}}>MARKERS ✦</div><div style={{fontSize:14,color:"#ffd700",fontFamily:"monospace"}}>{markers>0?"✦".repeat(Math.min(markers,10)):0}{markers>10?` +${markers-10}`:""}</div></div>
        <div style={{background:"rgba(0,40,30,.3)",border:"1px solid #00ff8822",borderRadius:6,padding:"6px 8px",fontSize:9,color:"#557788",lineHeight:1.7,fontFamily:"monospace"}}><div>←→ Move</div><div>↑ Rotate</div><div>↓ Soft Drop</div><div>Space Hard Drop</div><div>P Pause</div></div>
        <div style={{background:"rgba(0,40,30,.4)",border:"1px solid #00ff8833",borderRadius:6,padding:"6px 8px",textAlign:"center"}}>

          <div style={{display:"flex",flexWrap:"wrap",justifyContent:"center",gap:3}}>
            {CREW.map(c=>{const found=unlockedCrew.find(u=>u.id===c.id);const F=FACES[c.id];return(<div key={c.id} style={{border:found?`2px solid ${CHAR_COLORS[c.id]}`:"2px solid #333",borderRadius:4,padding:1,background:found?"rgba(255,255,255,.05)":"rgba(0,0,0,.3)",opacity:found?1:.3}} title={found?`${c.name} — ${c.title}`:"???"}>
              {found?<F size={22}/>:<div style={{width:22,height:22,display:"flex",alignItems:"center",justifyContent:"center",color:"#555",fontSize:10,fontWeight:700}}>?</div>}
            </div>)})}
          </div>
          <div style={{fontSize:7,color:"#556",marginTop:3}}>Collect them all!</div>
        </div>
        {nextToUnlock&&NextFace&&encMsg&&<div style={{background:"rgba(255,255,255,.04)",border:"1px solid rgba(255,255,255,.08)",borderRadius:6,padding:"6px 8px",display:"flex",gap:6,alignItems:"center"}}>
          <div style={{flexShrink:0,opacity:.7}}><NextFace size={24}/></div>
          <div style={{fontSize:9,color:"#99aabb",lineHeight:1.3,fontStyle:"italic"}} key={encIdx}>"{encMsg}"</div>
        </div>}
      </div>
    </div>
    <div style={{width:"100%",maxWidth:470,minHeight:60,marginTop:10,display:"flex",justifyContent:"center",zIndex:10}}>
      {tipVisible&&!paused&&!transiting&&<TipPopup tip={TIPS[tipIndex]} onClose={closeTip}/>}
    </div>
    <div style={{fontSize:8,color:"#334455",marginTop:6,zIndex:2,letterSpacing:.5}}>© 2001 FunTime Studios International · CEC Approved Recreation Software</div>
  </div>);
